# Solution fájl létrehozása
dotnet new sln -n ImperaBook

=======================================
|   DOTNET PROJECT 
=======================================

dotnet new classlib -lang "C#" -n ImperaBook.Domain
dotnet sln add ImperaBook.Domain

=======================================
|    TEST PROJECT
=======================================

dotnet new xunit -o ImperaBook.Domain.Tests
dotnet sln add ImperaBook.Domain


=======================================
|   MINIMAL WEB API
=======================================

dotnet new web -o ImperaBook.Api

=======================================
|   ADD REFERENCE 
=======================================

dotnet add reference ImperaBook.Domain/ImperaBook.Domain.fsproj

=======================================
|   ADD NUGET PACKAGE 
=======================================

dotnet add package Microsoft.EntityFrameworkCore.PostgreSQL


=======================================
|    EFCORE
=======================================
dotnet new tool-manifest          # if you don't already have .config/dotnet-tools.json
dotnet tool install dotnet-ef

============================
MIGRATION
============================
dotnet ef migrations add InitialCreate \
  --project src/ImperaBook.Infrastructure.Data \
  --startup-project src/ImperaBook.Infrastructure.Data

============================
UPDATE
============================
dotnet ef database update \
  --project src/ImperaBook.Infrastructure.Data \
  --startup-project src/ImperaBook.Infrastructure.Data

============================
VERIFY
============================
docker compose exec db psql -U imperabook -d imperabook -c "\dt"


============================
RESET DATABASE
============================
ResetDatabaseOnStartup=true dotnet run --project src/ImperaBook.Api
dotnet run --project backend/ImperaBook.Api --launch-profile reset-seed

============================
PSQL
============================

docker compose exec db psql -U imperabook -d imperabook
docker compose exec db psql -U alertme -d alertme

============================
DOCKER COMPOSE
============================
docker compose up -d
# wait for healthy (a few seconds); check with:
docker compose ps


==============================
DOCKER MANAGEMENT
==============================
docker stop alertme-db
docker stats
